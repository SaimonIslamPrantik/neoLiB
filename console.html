Supabase SQL Setup Script

Run this once in Supabase → SQL Editor. It creates your shelves, books, and password-protected database functions (default password: 1234). 

create extension if not exists pgcrypto;

create table if not exists public.shelves (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  code text not null default 'SH-01',
  description text not null default '',
  accent text not null default 'lavender',
  created_at timestamptz not null default now()
);

create table if not exists public.books (
  id uuid primary key default gen_random_uuid(),
  shelf_id uuid not null references public.shelves(id) on delete cascade,
  title text not null,
  author text not null default '',
  format text not null default 'PDF',
  thumbnail_url text not null default '',
  download_url text not null,
  created_at timestamptz not null default now()
);

create table if not exists public.console_config (
  id integer primary key default 1 check (id = 1),
  password_hash text not null,
  updated_at timestamptz not null default now()
);

create index if not exists idx_books_shelf_id on public.books(shelf_id);

alter table public.shelves enable row level security;
alter table public.books enable row level security;
alter table public.console_config enable row level security;

drop policy if exists "Allow public read shelves" on public.shelves;
create policy "Allow public read shelves"
  on public.shelves for select to anon, authenticated using (true);

drop policy if exists "Allow public read books" on public.books;
create policy "Allow public read books"
  on public.books for select to anon, authenticated using (true);

create or replace function public._neolib_seed_pw()
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  insert into public.console_config (id, password_hash)
  values (1, crypt('1234', gen_salt('bf')))
  on conflict (id) do nothing;
end;
$$;
select public._neolib_seed_pw();
drop function public._neolib_seed_pw();

create or replace function public.verify_console_password(p_password text)
returns boolean
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  return exists (
    select 1 from public.console_config
    where id = 1 and password_hash = crypt(coalesce(p_password, ''), password_hash)
  );
end;
$$;

create or replace function public.change_console_password(p_old_password text, p_new_password text)
returns boolean
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  if not public.verify_console_password(p_old_password) then
    raise exception 'Current password is incorrect';
  end if;
  if length(trim(coalesce(p_new_password, ''))) = 0 then
    raise exception 'New password cannot be empty';
  end if;

  update public.console_config
  set password_hash = crypt(p_new_password, gen_salt('bf')),
      updated_at = now()
  where id = 1;
  return true;
end;
$$;

create or replace function public.admin_save_shelf(
  p_password text,
  p_id uuid,
  p_name text,
  p_code text,
  p_description text,
  p_accent text
)
returns public.shelves
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_row public.shelves;
begin
  if not public.verify_console_password(p_password) then
    raise exception 'Unauthorized: invalid console password';
  end if;

  if p_id is null then
    insert into public.shelves (name, code, description, accent)
    values (trim(p_name), coalesce(nullif(trim(p_code), ''), 'SH-01'), coalesce(trim(p_description), ''), coalesce(nullif(trim(p_accent), ''), 'lavender'))
    returning * into v_row;
  else
    update public.shelves
    set name = trim(p_name),
        code = coalesce(nullif(trim(p_code), ''), 'SH-01'),
        description = coalesce(trim(p_description), ''),
        accent = coalesce(nullif(trim(p_accent), ''), 'lavender')
    where id = p_id
    returning * into v_row;
  end if;

  return v_row;
end;
$$;

create or replace function public.admin_delete_shelf(p_password text, p_id uuid)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  if not public.verify_console_password(p_password) then
    raise exception 'Unauthorized: invalid console password';
  end if;
  delete from public.shelves where id = p_id;
end;
$$;

create or replace function public.admin_save_book(
  p_password text,
  p_id uuid,
  p_shelf_id uuid,
  p_title text,
  p_author text,
  p_format text,
  p_thumbnail_url text,
  p_download_url text
)
returns public.books
language plpgsql
security definer
set search_path = public, extensions
as $$
declare
  v_row public.books;
begin
  if not public.verify_console_password(p_password) then
    raise exception 'Unauthorized: invalid console password';
  end if;

  if p_id is null then
    insert into public.books (shelf_id, title, author, format, thumbnail_url, download_url)
    values (p_shelf_id, trim(p_title), coalesce(trim(p_author), ''), upper(coalesce(nullif(trim(p_format), ''), 'PDF')), coalesce(trim(p_thumbnail_url), ''), trim(p_download_url))
    returning * into v_row;
  else
    update public.books
    set shelf_id = p_shelf_id,
        title = trim(p_title),
        author = coalesce(trim(p_author), ''),
        format = upper(coalesce(nullif(trim(p_format), ''), 'PDF')),
        thumbnail_url = coalesce(trim(p_thumbnail_url), ''),
        download_url = trim(p_download_url)
    where id = p_id
    returning * into v_row;
  end if;

  return v_row;
end;
$$;

create or replace function public.admin_delete_book(p_password text, p_id uuid)
returns void
language plpgsql
security definer
set search_path = public, extensions
as $$
begin
  if not public.verify_console_password(p_password) then
    raise exception 'Unauthorized: invalid console password';
  end if;
  delete from public.books where id = p_id;
end;
$$;

grant execute on function public.admin_delete_shelf(text, uuid) to anon, authenticated;
grant execute on function public.admin_save_book(text, uuid, uuid, text, text, text, text, text) to anon, authenticated;
grant execute on function public.admin_delete_book(text, uuid) to anon, authenticated;
