Supabase SQL Setup Script

Run this in Supabase → SQL Editor. Replace your-admin-email@example.com with your Supabase Auth user email. 

create extension if not exists pgcrypto;

create table if not exists public.shelves (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  code text not null default 'SH-01',
  description text not null default '',
  accent text not null default 'lavender',
  created_at timestamptz not null default now()
);

create table if not exists public.books (
  id uuid primary key default gen_random_uuid(),
  shelf_id uuid not null references public.shelves(id) on delete cascade,
  title text not null,
  author text not null default '',
  format text not null default 'PDF',
  thumbnail_url text not null default '',
  download_url text not null,
  created_at timestamptz not null default now()
);

create index if not exists idx_books_shelf_id on public.books(shelf_id);

alter table public.shelves enable row level security;
alter table public.books enable row level security;

create policy "Allow public read shelves"
  on public.shelves for select to anon, authenticated using (true);

create policy "Allow public read books"
  on public.books for select to anon, authenticated using (true);

create policy "Admin write shelves"
  on public.shelves for all to authenticated
  using ((auth.jwt() ->> 'email') = 'your-admin-email@example.com')
  with check ((auth.jwt() ->> 'email') = 'your-admin-email@example.com');

create policy "Admin write books"
  on public.books for all to authenticated
  using ((auth.jwt() ->> 'email') = 'your-admin-email@example.com')
  with check ((auth.jwt() ->> 'email') = 'your-admin-email@example.com');
